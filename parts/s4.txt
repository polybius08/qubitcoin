    def height(self) -> int:
        return len(self.blocks) - 1

        return self.blocks[-1]

    def bits(self) -> int:
        if len(self.blocks) < RETARGET_EVERY or len(self.blocks) % RETARGET_EVERY != 0:
            return self.tip()["header"]["bits"]
        span = self.tip()["header"]["time"] - self.blocks[-RETARGET_EVERY]["header"]["time"]
        span = max(TARGET_SPACING * RETARGET_EVERY // 4, min(span, TARGET_SPACING * RETARGET_EVERY * 4))
        target = max(1, compact_to_target(self.tip()["header"]["bits"]) * span // (TARGET_SPACING * RETARGET_EVERY))
        floor = compact_to_target(GENESIS_BITS)
        return target_to_compact(max(target, floor) if target < floor else target)
