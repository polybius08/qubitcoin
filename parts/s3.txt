def txid_of(tx: dict) -> str:
    body = {
        "version": tx["version"],
        "inputs": [{"txid": i["txid"], "vout": i["vout"], "pubkey": i.get("pubkey", ""), "height": i.get("height", 0), "message": i.get("message", "")} for i in tx["inputs"]],
        "outputs": tx["outputs"],
        "locktime": tx["locktime"],
    }
    return dhash_hex(json.dumps(body, separators=(",", ":"), sort_keys=True).encode())


def sighash(tx: dict) -> bytes:
    body = {
        "version": tx["version"],
        "inputs": [{"txid": i["txid"], "vout": i["vout"]} for i in tx["inputs"]],
        "outputs": tx["outputs"],
        "locktime": tx["locktime"],
    }
    return sha256(json.dumps(body, separators=(",", ":"), sort_keys=True).encode())


def header_bytes(block: dict) -> bytes:
    h = block["header"]
    return json.dumps({
        "version": h["version"], "prev": h["prev"], "merkle": h["merkle"],
        "time": h["time"], "bits": h["bits"], "nonce": h["nonce"], "magic": MAGIC,
    }, separators=(",", ":"), sort_keys=True).encode()


def block_hash(block: dict) -> str:
    return dhash_hex(header_bytes(block))
